import { Card, CardGrid, FileTree } from "@astrojs/starlight/components";

<CardGrid stagger>
  <Card title="Install without hassle" icon="download">
    Just [add one file](/using/install) to your game install to install GDPatch. No game decompiling required. Delete it any time to go back to a vanilla game install.
  </Card>

  <Card title="Works everywhere" icon="code-branch">
    GDPatch supports all major versions of Godot 4.x, and runs on Windows, macOS, and Linux. If you're playing a Godot game, GDPatch probably works with it out of the box.
  </Card>

  <Card title="Add mods anytime" icon="puzzle">
    Mods are loaded at runtime and never touch your game files. Install mods by simply unzipping them in the `mods` folder.

    <FileTree>
      - GDPatch
        - mods
          - `Mod A/`
          - `Mod B/`
          - `Mod C/`
      - game.exe
    </FileTree>
  </Card>

  <Card title="Patch literally anything" icon="pencil">
    Edit any line of your target game's source code using GDPatch's [Lua patcher system](/modding/patcher). Increase your compatibility with game updates and conflicting mods.

    ```lua
    GDPatch.patch_script_as_text(
      "Scripts/Player.gdc",
      function(ctx, src)
        return src:gsub(
          "const MAX_HEALTH = 10",
          "const MAX_HEALTH = 100"
        )
      end
    )
    ```
  </Card>
</CardGrid>
